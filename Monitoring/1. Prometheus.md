```
sudo useradd --no-create-home --shell /bin/false prometheus    #  Юзер без home и без shell — принцип минимальных привилегий: даже если прометей взломают, гадёнышу некуда будет пойти.
```

```
cd /tmp
PROM_VERSION=$(curl -sL https://api.github.com/repos/prometheus/prometheus/releases/latest | grep -oP '"tag_name": "\K[^"]+')
echo "Ставим: $PROM_VERSION"
wget -q "https://github.com/prometheus/prometheus/releases/download/${PROM_VERSION}/prometheus-${PROM_VERSION#v}.linux-amd64.tar.gz"
tar xf prometheus-${PROM_VERSION#v}.linux-amd64.tar.gz
cd prometheus-${PROM_VERSION#v}.linux-amd64
ls -la    # увидишь: prometheus, promtool, prometheus.yml, consoles/, console_libraries/
```

```
sudo cp prometheus promtool /usr/local/bin/
sudo cp prometheus.yml /etc/prometheus/
sudo chown -R prometheus:prometheus /etc/prometheus /var/lib/prometheus
promtool --version    # работает без sudo? отлично, бинарники на месте


# Бинарники оставляем root:root — исполняемым файлам не нужно владение prometheus, ему нужны только конфиг и data-директория.
```
