
- 1. делаем папку

```bash
mkdir -p ~/monitoring-course/03-node-exporter
cd ~/monitoring-course/03-node-exporter
```

- 2. системный юзер

```bash
sudo useradd --no-create-home --shell /bin/false node_exporter
```

- 3. качаем бинарник

```bash
cd /tmp
NE_VERSION=$(curl -sL "https://api.github.com/repos/prometheus/node_exporter/releases/latest" | grep -oP '"tag_name": "\K[^"]+')
echo "Ставим Node Exporter $NE_VERSION"
wget -q "https://github.com/prometheus/node_exporter/releases/download/${NE_VERSION}/node_exporter-${NE_VERSION#v}.linux-amd64.tar.gz"
tar xf node_exporter-${NE_VERSION#v}.linux-amd64.tar.gz
cd node_exporter-${NE_VERSION#v}.linux-amd64
ls -la    # должен быть только node_exporter (бинарник)
```

- 4. Раскладываем

```bash
sudo cp node_exporter /usr/local/bin
sudo chown node_exporter:node_exporter /usr/local/bin/node_exporter
node_exporter --version  # проверка бинарника, что он рабочий
```
